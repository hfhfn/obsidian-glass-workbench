# ⚡ 交互工作台

> [!summary] 这个页面是真交互的
> - **数字卡片**：点击直接跳到对应 MOC / 导航页
> - **筛选 chips**：按主题过滤下方收录列表
> - **搜索框**：输入标题关键词即时过滤
> - **列表行**：点击打开笔记；卡片支持指针跟随高光
>
> 配色：18 套（US-22～25 为炫彩）；打开 **设置 → Style Settings → 炫彩渐变** 后，本页头图、数字和卡片边框会变成多色流光。
> 
> 依赖 DataviewJS（设置 → Dataview → Enable JavaScript queries）。WebGL 流体版仍是独立 HTML，浏览器打开效果最佳：[[00_系统/工作台-玻璃展厅.html|🌊 玻璃展厅]]

```dataviewjs
(() => {
// ===== 数据源 =====
const SRC = '"AI大模型" or "AI" or "AIGC" or "CS(计科)" or "工作" or "应用&工具笔记" or "文学&认知" or "问题&资料收集"';
const pages = dv.pages(SRC).where(p => p.file.path !== dv.current().file.path);
const ts = p => {
  const c = p.file.ctime;
  if (!c) return 0;
  return typeof c.toMillis === "function" ? c.toMillis() : (c.ts || 0);
};
const top = p => (p.file.folder || "").split("/")[0];
const has = (p, prefix) => (p.file.path || "").startsWith(prefix);
const now = Date.now();
const mStart = new Date();
mStart.setDate(1);
mStart.setHours(0, 0, 0, 0);
const cnt = fn => pages.filter(fn).length;
const inMonth = p => ts(p) >= mStart.getTime();

// ===== 指标卡片 =====
const topics = [
  { label: "全库笔记", value: pages.length, sub: `本月 +${cnt(inMonth)}`, target: "00_系统/知识库导航页" },
  { label: "AI 大模型", value: cnt(p => has(p, "AI大模型/")), sub: `本月 +${cnt(p => has(p, "AI大模型/") && inMonth(p))}`, target: "AI大模型/AI大模型-MOC" },
  { label: "Agent", value: cnt(p => has(p, "AI大模型/Agent/")), sub: `本月 +${cnt(p => has(p, "AI大模型/Agent/") && inMonth(p))}`, target: "AI大模型/Agent/Agent演进全过程-Loop工程到Graph工程" },
  { label: "RAG", value: cnt(p => has(p, "AI大模型/rag/")), sub: `本月 +${cnt(p => has(p, "AI大模型/rag/") && inMonth(p))}`, target: "AI大模型/rag/RAG系统核心技术与实现" },
  { label: "面试题", value: cnt(p => has(p, "工作/面试收集/")), sub: `本月 +${cnt(p => has(p, "工作/面试收集/") && inMonth(p))}`, target: "工作/面试收集/面试收集-MOC" },
  { label: "工作", value: cnt(p => has(p, "工作/")), sub: `本月 +${cnt(p => has(p, "工作/") && inMonth(p))}`, target: "工作/工作-MOC" },
  { label: "本月新增", value: cnt(inMonth), sub: "按收录时间", target: "00_系统/知识库导航页" },
  { label: "近 7 天", value: cnt(p => ts(p) >= now - 7 * 864e5), sub: "最近收录", target: "00_系统/知识库导航页" }
];

// ===== DOM =====
const root = dv.container;
if (root.dataset.wbMounted === "1") return;
root.dataset.wbMounted = "1";
root.innerHTML = "";
root.classList.add("wb-dash");
const mk = (tag, cls, parent = root) => {
  const el = document.createElement(tag);
  if (cls) el.className = cls;
  parent.appendChild(el);
  return el;
};

// ===== 玻璃头图（CSS 极光，不依赖 WebGL） =====
const hero = mk("div", "wb-hero");
mk("div", "wb-hero-title", hero).textContent = "LIVE · 玻璃工作台";
mk("div", "wb-hero-sub", hero).textContent = `${pages.length} 篇笔记 · 鼠标移动可改变光晕`;
hero.addEventListener("pointermove", e => {
  const r = hero.getBoundingClientRect();
  hero.style.setProperty("--hx", `${((e.clientX - r.left) / Math.max(r.width, 1)) * 100}%`);
  hero.style.setProperty("--hy", `${((e.clientY - r.top) / Math.max(r.height, 1)) * 100}%`);
});
hero.addEventListener("pointerleave", () => {
  hero.style.setProperty("--hx", "30%");
  hero.style.setProperty("--hy", "35%");
});

// 卡片
const grid = mk("div", "wb-cards");
topics.forEach(t => {
  const card = mk("div", "wb-card", grid);
  card.tabIndex = 0;
  card.setAttribute("role", "link");
  card.title = `打开：${t.target}`;
  mk("div", "wb-label", card).textContent = t.label;
  mk("div", "wb-value", card).textContent = String(t.value);
  mk("div", "wb-sub", card).textContent = t.sub;
  const open = () => dv.app.workspace.openLinkText(t.target, "", false);
  card.addEventListener("click", open);
  card.addEventListener("keydown", e => {
    if (e.key === "Enter" || e.key === " ") {
      e.preventDefault();
      open();
    }
  });
  card.addEventListener("pointermove", e => {
    const r = card.getBoundingClientRect();
    card.style.setProperty("--mx", `${e.clientX - r.left}px`);
    card.style.setProperty("--my", `${e.clientY - r.top}px`);
  });
});

// 筛选栏
const bar = mk("div", "wb-toolbar");
const names = ["全部", "AI大模型", "Agent", "RAG", "AI", "AIGC", "CS(计科)", "工作", "应用&工具笔记", "文学&认知", "问题&资料收集"];
const FILTERS = {
  "全部": () => true,
  "AI大模型": p => top(p) === "AI大模型",
  "Agent": p => has(p, "AI大模型/Agent/"),
  "RAG": p => has(p, "AI大模型/rag/"),
  "AI": p => top(p) === "AI",
  "AIGC": p => top(p) === "AIGC",
  "CS(计科)": p => top(p) === "CS(计科)",
  "工作": p => top(p) === "工作",
  "应用&工具笔记": p => top(p) === "应用&工具笔记",
  "文学&认知": p => top(p) === "文学&认知",
  "问题&资料收集": p => top(p) === "问题&资料收集"
};
let active = "全部";
let query = "";
const chipEls = {};
names.forEach(name => {
  const b = mk("button", "wb-chip" + (name === "全部" ? " is-active" : ""), bar);
  b.type = "button";
  b.textContent = name;
  b.onclick = () => {
    active = name;
    Object.entries(chipEls).forEach(([k, el]) => el.classList.toggle("is-active", k === name));
    render();
  };
  chipEls[name] = b;
});

const search = mk("input", "wb-search", bar);
search.type = "search";
search.placeholder = "搜索标题…";
search.oninput = () => {
  query = search.value.trim().toLowerCase();
  render();
};

// 列表
const listTitle = mk("div", "wb-list-title");
const titleLeft = mk("span", "", listTitle);
const titleRight = mk("span", "", listTitle);
const list = mk("div", "wb-list");

function render() {
  list.innerHTML = "";
  let items = pages.filter(p => ts(p) >= now - 183 * 864e5);
  if (items.length < 10) items = pages.filter(p => ts(p) > 0);
  items = items.filter(FILTERS[active]);
  if (query) items = items.filter(p => (p.file.name || "").toLowerCase().includes(query));
  items = [...items].sort((a, b) => ts(b) - ts(a));

  titleLeft.textContent = `${active === "全部" ? "最近收录" : active + " · 最近收录"} · ${items.length} 篇`;
  titleRight.textContent = query ? `筛选：“${query}”` : "点击行打开笔记";

  const slice = items.slice(0, 80);
  if (!slice.length) {
    mk("div", "wb-empty", list).textContent = "没有匹配的笔记";
    return;
  }

  slice.forEach(p => {
    const row = mk("div", "wb-row", list);
    row.title = p.file.path;
    mk("div", "wb-row-name", row).textContent = p.file.name;
    mk("div", "wb-row-folder", row).textContent = p.file.folder || "根目录";
    const c = p.file.ctime;
    mk("div", "wb-row-date", row).textContent =
      c && typeof c.toFormat === "function" ? c.toFormat("yyyy-MM-dd") : "";
    row.onclick = () => dv.app.workspace.openLinkText(p.file.path, "", false);
  });
}

render();

})();
```
